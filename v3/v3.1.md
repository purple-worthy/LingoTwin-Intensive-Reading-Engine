### 二、 终极电子书渲染引擎 (`ielts_reader_engine_v3.html`)

这个 HTML 文件就是你**永久复用的“唯一骨架”**。
以后你让任何 AI 读文章（无论是让 AI 从网上抓取、读本地文件，还是你直接粘贴文章发给 AI），**AI 只要吐出一段 JSON，你点网页顶部的 `➕ 导入 / 粘贴新文章` 把 JSON 粘进去，网页立刻自动渲染并加入左侧目录**（还会自动保存在浏览器 `localStorage` 里，刷新网页也不会丢）！

复制保存为 `ielts_reader_engine_v3.html`，用浏览器打开即可使用：

function renderDrawer() {

const list = document.getElementById('drawerList');
list.innerHTML = articlesDB.map((art, idx) => `
<button class="drawer-item ${idx === currentIndex ? 'active' : ''}" onclick="selectArticle(${idx})">

<small>${art.titleEn}</small>
</button>
`).join('');
}

function renderArticle(art) {
if (!art) return;
const parasHtml = art.paragraphs.map(p => {
const rubies = p.tokens.map(t => {
const tierClass = `tier-${t.tier || 'normal'}`;
return `<ruby class="${tierClass}" onclick="speakText('${t.text.replace(/'/g, "\\'")}')">${t.text}<rt>${t.gloss || ''}</rt></ruby>`;
}).join(' ');
return `
<p class="gloss-paragraph">${rubies}</p>
<div class="full-translation">${p.translation}</div>
`;
}).join('');

const vocabHtml = art.vocabulary.map(v => {
const defsHtml = v.defs.map(d => `<div><span class="pos-tag">${d.pos}</span> ${d.meaning}</div>`).join('');
return `
<div class="vocab-item" data-level="${v.level || 'core'}">
<div class="vocab-head" onclick="speakText('${v.word.replace(/'/g, "\\'")}')">
<span><span class="vocab-word">${v.word}</span> <span class="vocab-ipa">${v.ipa || ''}</span></span>
<span style="font-size:11px;opacity:0.5;">🔊</span>
</div>
<div class="vocab-def">${defsHtml}</div>
</div>
`;
}).join('');

document.getElementById('readerRoot').innerHTML = `
<header class="top-banner">
<div class="badge-box">雅思</div>
<div class="banner-titles">
<h1>短文精读与全词解析真经</h1>
<div class="chapter-info">${art.category || '精选阅读单元'}</div>
</div>
</header>
<div class="meta-bar">

</main>
<aside class="right-column">

</div>

function speakText(text) {
if (!('speechSynthesis' in window)) return;
window.speechSynthesis.cancel();
const clean = text.replace(/[.,!?;:"'()]/g, '').trim();
if (!clean) return;
const u = new SpeechSynthesisUtterance(clean);
u.lang = 'en-US';
u.rate = 0.92;
window.speechSynthesis.speak(u);
}

function importJsonArticle() {
const raw = document.getElementById('jsonInput').value.trim();
if (!raw) return alert('请先粘贴 AI 生成的 JSON 数据！');
try {
const cleaned = raw.replace(/^```json\s*/i, '').replace(/```\s*$/, '');
const obj = JSON.parse(cleaned);
if (!obj.titleEn || !obj.paragraphs) throw new Error('JSON 缺少 titleEn 或 paragraphs 字段');
articlesDB.push(obj);
localStorage.setItem('ielts_articles_v3', JSON.stringify(articlesDB));
currentIndex = articlesDB.length - 1;
renderApp();
closeImportModal();
document.getElementById('jsonInput').value = '';
} catch (e) {
alert('JSON 格式解析失败，请检查是否完整复制：\n' + e.message);
}
}

});
localStorage.setItem('ielts_articles_v3', JSON.stringify(articlesDB));
currentIndex = articlesDB.length - 1;
renderApp();
closeImportModal();
document.getElementById('jsonInput').value = '';
}
