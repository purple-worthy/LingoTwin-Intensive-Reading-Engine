async function handleMultiFormatEbook(input) {
const file = input.files[0];
if (!file) return;
const statusEl = document.getElementById('ebookParseStatus');
const ext = file.name.split('.').pop().toLowerCase();
const bookName = file.name.replace(/\.[^/.]+$/, '');
statusEl.textContent = `⏳ 正在解析《${bookName}》(${ext.toUpperCase()})，请稍候...`;

try {
let extractedChapters = [];

if (ext === 'epub') {
extractedChapters = await parseEpubFile(file);
} else if (ext === 'pdf') {
extractedChapters = await parsePdfFile(file);
} else if (ext === 'mobi' || ext === 'azw' || ext === 'azw3') {
extractedChapters = await parseMobiAzwFile(file);
} else {
const text = await file.text();
extractedChapters = splitPlainTextToChapters(text);
}

if (!extractedChapters.length) {
throw new Error('未能提取到有效文本内容（若是扫描版图片 PDF 或加密 DRM 电子书，需先进行 OCR 或解密）。');
}

extractedChapters.forEach((ch, idx) => {
const paras = ch.paragraphs.map(line => ({
tokens: line.split(/\s+/).filter(Boolean).map(w => ({ text: w, gloss: "", tier: w.length > 6 ? "red" : "normal" })),
translation: "（已导入电子书原文：点击顶部 [📋 复制本章给 AI 解析] 即可生成逐词注释与词表）"
}));
window.ReaderApp.articles.push({
id: `ebook-${Date.now()}-${idx}`,
isUnparsed: true,
category: `${ext.toUpperCase()} 电子书 · 《${bookName}》`,

titleCn: `第 ${idx + 1} 章 · ${ch.title || ''}`,

});
});

window.ReaderApp.currentIndex = window.ReaderApp.articles.length - extractedChapters.length;
localStorage.setItem('ielts_ebook_v5', JSON.stringify(window.ReaderApp.articles));
renderAll();
statusEl.textContent = `✅ 成功解析《${bookName}》，共提取 ${extractedChapters.length} 个章节！`;

// 1. EPUB 解析器（基于 JSZip 解压读取 OPF 清单与 XHTML 正文）
async function parseEpubFile(file) {
const zip = await JSZip.loadAsync(file);
const containerXml = await zip.file('META-INF/container.xml').async('string');
const opfPath = containerXml.match(/full-path="([^"]+)"/)[1];
const opfDir = opfPath.includes('/') ? opfPath.substring(0, opfPath.lastIndexOf('/') + 1) : '';
const opfXml = await zip.file(opfPath).async('string');
const parser = new DOMParser();
const opfDoc = parser.parseFromString(opfXml, 'text/xml');

const manifest = {};
opfDoc.querySelectorAll('manifest > item').forEach(item => {
manifest[item.getAttribute('id')] = opfDir + item.getAttribute('href');
});

const chapters = [];
const spineItems = opfDoc.querySelectorAll('spine > itemref');
for (const ref of spineItems) {
const href = manifest[ref.getAttribute('idref')];
if (!href || !zip.file(href)) continue;
const htmlStr = await zip.file(href).async('string');
const doc = parser.parseFromString(htmlStr, 'text/html');
const titleEl = doc.querySelector('h1, h2, h3, title');
const title = titleEl ? titleEl.textContent.trim() : '';
const paras = Array.from(doc.querySelectorAll('p'))
.map(p => p.textContent.replace(/\s+/g, ' ').trim())
.filter(t => t.length > 20);
if (paras.length > 0) chapters.push({ title: title.slice(0, 60), paragraphs: paras });
}
return chapters;
}

// 2. PDF 解析器（基于 PDF.js 逐页提取并每 5 页合并为一个阅读单元）
async function parsePdfFile(file) {
const buf = await file.arrayBuffer();
const pdf = await pdfjsLib.getDocument({ data: buf }).promise;
const chapters = [];
const pagesPerChapter = 4;
for (let i = 1; i <= pdf.numPages; i += pagesPerChapter) {
let chunkParas = [];
const endPage = Math.min(i + pagesPerChapter - 1, pdf.numPages);
for (let p = i; p <= endPage; p++) {
const page = await pdf.getPage(p);
const content = await page.getTextContent();
const pageText = content.items.map(item => item.str).join(' ').replace(/\s+/g, ' ').trim();
if (pageText.length > 20) chunkParas.push(pageText);
}
if (chunkParas.length) {
chapters.push({ title: `Pages ${i} - ${endPage}`, paragraphs: chunkParas });
}
}
return chapters;
}

// 3. MOBI / AZW / AZW3 二进制文本流提取器
async function parseMobiAzwFile(file) {
const buf = await file.arrayBuffer();
const decoder = new TextDecoder('utf-8', { fatal: false });
const rawStr = decoder.decode(new Uint8Array(buf));
// 提取其中的 HTML 段落块或连续英文可读文本
const htmlMatches = rawStr.match(/<p[^>]*>[\s\S]*?<\/p>/gi);
if (htmlMatches && htmlMatches.length > 5) {
const cleanParas = htmlMatches
.map(h => h.replace(/<[^>]+>/g, ' ').replace(/\s+/g, ' ').trim())
.filter(t => t.length > 25 && /[a-zA-Z]{3,}/.test(t));
return splitParagraphsIntoChapters(cleanParas, 15);
}
// 回退：提取连续英文句段
const asciiBlocks = rawStr.match(/[A-Za-z0-9 ,;:'"?!.\-()]{80,}/g) || [];
return splitParagraphsIntoChapters(asciiBlocks.map(s => s.trim()), 12);
}

function splitPlainTextToChapters(text) {
const blocks = text.split(/\n(?=(?:Chapter|CHAPTER)\s+\d+|^[IVXLC]+\.\s)/m).filter(c => c.trim().length > 40);
return blocks.map(b => {
const lines = b.trim().split(/\n+/).map(l => l.trim()).filter(Boolean);
return { title: lines[0].slice(0, 60), paragraphs: lines.slice(1).length ? lines.slice(1) : lines };
});
}

function splitParagraphsIntoChapters(paras, size) {
const res = [];
for (let i = 0; i < paras.length; i += size) {
res.push({ title: `Section ${Math.floor(i / size) + 1}`, paragraphs: paras.slice(i, i + size) });
}
return res;
}
