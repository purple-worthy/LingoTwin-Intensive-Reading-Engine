function formatTime(ts) {
const d = new Date(ts);
return `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')} ${String(d.getHours()).padStart(2,'0')}:${String(d.getMinutes()).padStart(2,'0')}`;

function getFolderDepth(folderId) {
let depth = 0;
let cur = allNodes.find(n => n.id === folderId);
while (cur && cur.parentId) {
depth++;
cur = allNodes.find(n => n.id === cur.parentId);
}
return cur ? depth + 1 : 0;
}

let maxNum = 0;
siblings.forEach(s => {
const m = s.title.match(/^lbook(\d+)$/i);
if (m) maxNum = Math.max(maxNum, parseInt(m[1], 10));
});
return `lbook${maxNum + 1}`;
}

function sanitizeText(str) {

async function handleMultiFormatEbook(input) {
const file = input.files[0];
if (!file) return;

const targetFolderId = document.getElementById('importTargetFolder').value || null;
const ext = file.name.split('.').pop().toLowerCase();
// 清理文件名中的网站广告后缀（如 (z-library.sk, 1lib.sk)）
let rawBookTitle = sanitizeText(file.name.replace(/\.[^/.]+$/, '').replace(/\([^)]*(?:z-lib|1lib|anna)[^)]*\)/gi, ''));
if (!rawBookTitle) rawBookTitle = generateDefaultLbookName(targetFolderId);

statusEl.style.color = '#0f766e';
statusEl.textContent = `⏳ 正在深度解析《${rawBookTitle}》(${ext.toUpperCase()})，自动拆分章节中...`;

try {
let bookMetaTitle = rawBookTitle;
let chapters = [];

if (ext === 'epub') {
const res = await parseEpubClean(file);
if (res.bookTitle && res.bookTitle.length > 1) bookMetaTitle = res.bookTitle;
chapters = res.chapters;
} else if (ext === 'pdf') {
chapters = await parsePdfClean(file);
} else if (ext === 'mobi' || ext === 'azw' || ext === 'azw3') {
chapters = await parseMobiAzwClean(file);
} else {
const txt = await file.text();
chapters = splitPlainTextToChapters(txt);
}

if (!chapters.length) {
throw new Error('未能提取到有效文本段落（若为加密 DRM 电子书或纯图片扫描 PDF，请先解密或 OCR）。');
}

const now = Date.now();
// 1. 创建电子书折叠容器节点（点击书名可展开查看所有拆分好的章节）
const bookNodeId = 'book-' + now;

type: 'book',
title: bookMetaTitle,

expanded: true
});

for (let i = 0; i < chapters.length; i++) {

if (i === 0) firstChapId = chapId;

parentId: bookNodeId,
type: 'chapter',
title: `${i + 1}. ${cleanChapTitle}`,
createdAt: now + i
});

gloss: "",
tier: w.length > 6 ? "red" : "normal"
})),
translation: "（未解析原文：可点击右侧 [📋 复制本段] 或顶部 [📋 复制本章给 AI 解析]）"
}));

id: chapId,
isUnparsed: true,
category: `《${bookMetaTitle}》`,
titleEn: cleanChapTitle,
titleCn: `第 ${i + 1} 章 · ${cleanChapTitle}`,

// 深度净化版 EPUB 解析器：读取 OPF 元数据 + NCX/NAV 目录映射 + UTF-8 安全解码 + 超长单文件自动二次按章节切分
async function parseEpubClean(file) {
const zip = await JSZip.loadAsync(file);
const parser = new DOMParser();

const containerFile = zip.file('META-INF/container.xml');
if (!containerFile) throw new Error('无效的 EPUB 结构（缺少 container.xml）');
const containerXml = await containerFile.async('string');
const opfPath = containerXml.match(/full-path="([^"]+)"/)[1];
const opfDir = opfPath.includes('/') ? opfPath.substring(0, opfPath.lastIndexOf('/') + 1) : '';
const opfXml = await zip.file(opfPath).async('string');
const opfDoc = parser.parseFromString(opfXml, 'text/xml');

// 提取真实书名
const titleNode = opfDoc.querySelector('metadata title, dc\\:title');
const bookTitle = titleNode ? sanitizeText(titleNode.textContent) : '';

const manifest = {};
let ncxHref = null;
opfDoc.querySelectorAll('manifest > item').forEach(item => {
const id = item.getAttribute('id');
const href = decodeURIComponent(item.getAttribute('href') || '');
const mediaType = item.getAttribute('media-type') || '';
manifest[id] = opfDir + href;
if (mediaType.includes('ncx') || href.endsWith('.ncx')) ncxHref = opfDir + href;
});

// 尝试从 toc.ncx 获取精准章节标题映射 (src -> label)
const tocTitleMap = {};
if (ncxHref && zip.file(ncxHref)) {
const ncxXml = await zip.file(ncxHref).async('string');
const ncxDoc = parser.parseFromString(ncxXml, 'text/xml');
ncxDoc.querySelectorAll('navPoint').forEach(np => {
const label = sanitizeText(np.querySelector('navLabel > text')?.textContent);
const srcRaw = np.querySelector('content')?.getAttribute('src') || '';
const srcClean = decodeURIComponent(srcRaw.split('#')[0]);
if (label && srcClean) tocTitleMap[opfDir + srcClean] = label;
});
}

const chapters = [];
const spineItems = opfDoc.querySelectorAll('spine > itemref');
for (const ref of spineItems) {
const href = manifest[ref.getAttribute('idref')];
if (!href || !zip.file(href)) continue;

// 强制使用 Uint8Array + TextDecoder('utf-8') 读取，彻底杜绝二进制乱码
const bytes = await zip.file(href).async('uint8array');
const htmlStr = new TextDecoder('utf-8').decode(bytes);
const doc = parser.parseFromString(htmlStr, 'text/html');

// 优先使用 NCX 目录标题，其次是文档内 h1~h3（排除乱码）
let chTitle = tocTitleMap[href] || '';
if (!chTitle) {
const hEl = doc.querySelector('h1, h2, h3, .chapter-title, .title');
if (hEl) chTitle = sanitizeText(hEl.textContent);
}

const paras = Array.from(doc.querySelectorAll('p, div.para'))
.map(p => sanitizeText(p.textContent))
.filter(t => t.length > 15 && /[a-zA-Z]{2,}/.test(t));

if (paras.length === 0) continue;

// 防止某些 EPUB 把整本书塞在同一个大 XHTML 文件里（若单文件超 2500 词，自动按小标题或每 18 段切分为阅读单元）
const totalWords = paras.join(' ').split(/\s+/).length;
if (totalWords > 2500) {
const chunkSize = 18;
for (let k = 0; k < paras.length; k += chunkSize) {
const slice = paras.slice(k, k + chunkSize);
chapters.push({
title: chTitle ? `${chTitle} (Part ${Math.floor(k/chunkSize)+1})` : `Section ${chapters.length + 1}`,
paragraphs: slice
});
}
} else {
chapters.push({
title: chTitle || `Chapter ${chapters.length + 1}`,
paragraphs: paras
});
}
}
return { bookTitle, chapters };
}

async function parsePdfClean(file) {
const buf = await file.arrayBuffer();
const pdf = await pdfjsLib.getDocument({ data: buf }).promise;
const chapters = [];
const pagesPerChapter = 3;
for (let i = 1; i <= pdf.numPages; i += pagesPerChapter) {
let chunkParas = [];
const endPage = Math.min(i + pagesPerChapter - 1, pdf.numPages);
for (let p = i; p <= endPage; p++) {
const page = await pdf.getPage(p);
const content = await page.getTextContent();
const pageText = sanitizeText(content.items.map(item => item.str).join(' '));
if (pageText.length > 20) chunkParas.push(pageText);
}
if (chunkParas.length) {
chapters.push({ title: `Pages ${i} - ${endPage}`, paragraphs: chunkParas });
}
}
return chapters;
}

async function parseMobiAzwClean(file) {
const buf = await file.arrayBuffer();
const rawStr = new TextDecoder('utf-8', { fatal: false }).decode(new Uint8Array(buf));
const htmlMatches = rawStr.match(/<p[^>]*>[\s\S]*?<\/p>/gi) || [];
const cleanParas = htmlMatches
.map(h => sanitizeText(h.replace(/<[^>]+>/g, ' ')))
.filter(t => t.length > 25 && /[a-zA-Z]{3,}/.test(t));
const chapters = [];
for (let i = 0; i < cleanParas.length; i += 15) {
chapters.push({ title: `Chapter ${Math.floor(i / 15) + 1}`, paragraphs: cleanParas.slice(i, i + 15) });
}
return chapters;
}

function splitPlainTextToChapters(text) {
const blocks = text.split(/\n(?=(?:Chapter|CHAPTER)\s+\d+|^[IVXLC]+\.\s)/m).filter(c => c.trim().length > 40);
return blocks.map((b, i) => {
const lines = b.trim().split(/\n+/).map(l => sanitizeText(l)).filter(Boolean);
return { title: lines[0].slice(0, 60) || `Chapter ${i + 1}`, paragraphs: lines.slice(1).length ? lines.slice(1) : lines };

window.ReaderApp = {
currentArticle: null,
getCurrentArticle() { return this.currentArticle; },
speak(text) {
if (!('speechSynthesis' in window)) return;
window.speechSynthesis.cancel();
const clean = text.replace(/[.,!?;:"'()—]/g, '').trim();
if (!clean) return;
const u = new SpeechSynthesisUtterance(clean);
u.lang = 'en-US'; u.rate = 0.92;
window.speechSynthesis.speak(u);
}
};

async function openArticleById(id) {

activeArticleId = id;

}

function copySinglePara(pIdx, includeTrans) {
const art = window.ReaderApp.currentArticle;
if (!art || !art.paragraphs[pIdx]) return;
const p = art.paragraphs[pIdx];
const enText = p.tokens.map(t => t.text).join(' ');
const payload = includeTrans ? `${enText}\n${p.translation}` : enText;
navigator.clipboard.writeText(payload).then(() => {
alert(includeTrans ? '已复制本段双语内容！' : '已复制本段英文原文，可直接发给 AI 解析！');
});
}

async function importJsonArticle() {
const raw = document.getElementById('jsonInput').value.trim();
if (!raw) return alert('请粘贴 AI 生成的 JSON 数据！');
const targetFolderId = document.getElementById('importTargetFolder').value || null;
try {
const obj = JSON.parse(raw.replace(/^```json\s*/i, '').replace(/```\s*$/, ''));
const cur = window.ReaderApp.currentArticle;
if (cur && cur.isUnparsed && activeArticleId) {
obj.id = activeArticleId;
await LocalDB.put('articles', obj);
await openArticleById(activeArticleId);
} else {
const newId = 'art-' + Date.now();
const title = obj.titleCn || obj.titleEn || generateDefaultLbookName(targetFolderId);
obj.id = newId;
await LocalDB.put('nodes', { id: newId, parentId: targetFolderId, type: 'article', title, createdAt: Date.now() });
await LocalDB.put('articles', obj);
await refreshDirectoryTree();
await openArticleById(newId);
}
closeImportModal();
document.getElementById('jsonInput').value = '';
} catch (e) { alert('JSON 格式有误：\n' + e.message); }
}

async function quickRawPreview() {
const raw = document.getElementById('jsonInput').value.trim();
if (!raw || raw.startsWith('{')) return alert('请先粘贴纯英文段落！');
const targetFolderId = document.getElementById('importTargetFolder').value || null;
const defaultTitle = generateDefaultLbookName(targetFolderId);
const newId = 'raw-' + Date.now();
const paras = raw.split(/\n+/).filter(Boolean).map(line => ({
tokens: line.split(/\s+/).map(w => ({ text: w, gloss: "", tier: w.length > 6 ? "red" : "normal" })),
translation: "（未解析原文：可点击右侧 [📋 复制英文] 发给 AI 获取全词释义）"
}));
await LocalDB.put('nodes', { id: newId, parentId: targetFolderId, type: 'article', title: defaultTitle, createdAt: Date.now() });
await LocalDB.put('articles', {
id: newId, isUnparsed: true, category: '独立文章精读',
titleEn: defaultTitle, titleCn: `${defaultTitle}（待 AI 解析）`,
wordCount: raw.split(/\s+/).length, paragraphs: paras, vocabulary: []
});
await refreshDirectoryTree();
await openArticleById(newId);
closeImportModal();
document.getElementById('jsonInput').value = '';
}
